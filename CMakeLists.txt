cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(ScannerRoute VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Visual Studio 2022 C++ tools for Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDK_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/Interceptor" CACHE PATH "Matching Interceptor Keyboard & Mouse SDK")
option(SCANNER_ASAN "Enable AddressSanitizer for model tests" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
add_library(ScannerCore App/Scanner.cpp)
target_include_directories(ScannerCore PUBLIC App)
add_executable(ScannerRoute WIN32 App/Main.cpp App/Session.cpp "${SDK_DIRECTORY}/Interceptor.cpp")
target_include_directories(ScannerRoute PRIVATE "${SDK_DIRECTORY}")
target_link_libraries(ScannerRoute PRIVATE ScannerCore comctl32 comdlg32 shell32 user32 gdi32 gdiplus advapi32 dwmapi uxtheme)
target_link_options(ScannerRoute PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_executable(ScannerTests Tests/ScannerTests.cpp)
target_link_libraries(ScannerTests PRIVATE ScannerCore)
if(SCANNER_ASAN)
  target_compile_options(ScannerCore PRIVATE /fsanitize=address /Zi)
  target_compile_options(ScannerTests PRIVATE /fsanitize=address /Zi)
  target_link_options(ScannerTests PRIVATE /INCREMENTAL:NO)
endif()
enable_testing()
add_test(NAME ScannerModel COMMAND ScannerTests)
